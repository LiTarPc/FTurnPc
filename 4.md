Сейчас `Orchestrator` жёстко знает про `*FreeturnEngine`. Вводим интерфейс — дальше оба движка

// backend/engine.go
type Engine interface {

### 6.2 `backend/engine_singbox.go`

```go
package backend

import (
	"bufio"
	"context"
	"fmt"
	"io"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"sync"
	"time"

	"github.com/wailsapp/wails/v2/pkg/runtime"
)

type SingboxEngine struct {
	appCtx context.Context
	mu     sync.Mutex
	wg     sync.WaitGroup

	cmd      *exec.Cmd
	cancel   context.CancelFunc
	exitChan chan struct{}

	onTray           func(connected bool, rx, tx int64, workers int32)
	onUnexpectedExit func(err error)

	userStopped bool
	started     bool
	statsStop   chan struct{}

	cfgPath   string
	secret    string
	clashPort int
}

func NewSingboxEngine(ctx context.Context, onTray func(bool, int64, int64, int32), onExit func(error)) *SingboxEngine {
	return &SingboxEngine{appCtx: ctx, onTray: onTray, onUnexpectedExit: onExit}
}

// getSingboxPath — по образцу getFreeturnPath: рядом с exe, в assets, в PATH.
func getSingboxPath() string {
	name := "sing-box"
	if goruntime.GOOS == "windows" {
		name = "sing-box.exe"
	}
	exe, _ := os.Executable()
	dir := filepath.Dir(exe)
	for _, p := range []string{
		filepath.Join(dir, name),
		filepath.Join(dir, "assets", "singbox", name),
		filepath.Join(configDir(), "core", name),
	} {
		if _, err := os.Stat(p); err == nil {
			return p
		}
	}
	if p, err := exec.LookPath(name); err == nil {
		return p
	}
	return filepath.Join(dir, name)
}

func (e *SingboxEngine) Start(p ConnectParams, prof *ProfileData) error {
	e.mu.Lock()
	defer e.mu.Unlock()
	if e.cmd != nil {
		return fmt.Errorf("already running")
	}

	exePath := getSingboxPath()
	if _, err := os.Stat(exePath); err != nil {
		return fmt.Errorf("ядро sing-box не найдено: %s (скачайте его в настройках)", exePath)
	}

	e.secret = newSecret()
	e.clashPort = firstPositive(prof.Sing.ClashPort, 9095)
	cfg, err := BuildSingboxConfig(prof, p, e.secret)
	if err != nil {
		return err
	}
	e.cfgPath = filepath.Join(singboxDir(), "config.json")
	if err := os.WriteFile(e.cfgPath, cfg, 0o600); err != nil {
		return fmt.Errorf("не удалось записать конфиг: %w", err)
	}
	if err := singboxCheck(exePath, e.cfgPath); err != nil {
		return err // текст ошибки уже человекочитаемый
	}

	ctx, cancel := context.WithCancel(context.Background())
	e.cancel = cancel
	e.exitChan = make(chan struct{})
	e.started = false
	e.userStopped = false

	args := []string{"run", "-c", e.cfgPath, "-D", singboxDir(), "--disable-color"}
	e.cmd = exec.CommandContext(ctx, exePath, args...)
	prepareProcessGroup(e.cmd) // hideWindow + CREATE_NEW_PROCESS_GROUP + job object

	stdout, err := e.cmd.StdoutPipe()
	if err != nil {
		cancel()
		return err
	}
	stderr, err := e.cmd.StderrPipe()
	if err != nil {
		cancel()
		return err
	}

	runtime.EventsEmit(e.appCtx, "log", "DEBUG", fmt.Sprintf("Launching sing-box: %s %v", exePath, args))
	if err := e.cmd.Start(); err != nil {
		cancel()
		e.cmd = nil
		return fmt.Errorf("не удалось запустить sing-box: %w", err)
	}
	runtime.EventsEmit(e.appCtx, "state_changed", "connecting", "")
	if e.onTray != nil {
		e.onTray(false, 0, 0, 0)
	}

	e.wg.Add(2)
	go e.parseLogs(stdout, p)
	go e.parseLogs(stderr, p)
	go e.waitExit()
	return nil
}

func (e *SingboxEngine) waitExit() {
	defer close(e.exitChan)
	err := e.cmd.Wait()
	e.mu.Lock()
	stopped := e.userStopped
	e.stopStatsLoopLocked()
	e.mu.Unlock()
	e.wg.Wait()

	runtime.EventsEmit(e.appCtx, "log", "INFO", fmt.Sprintf("Сессия sing-box завершена (err: %v)", err))
	if stopped {
		runtime.EventsEmit(e.appCtx, "state_changed", "disconnected", "")
	} else {
		runtime.EventsEmit(e.appCtx, "state_changed", "connecting", "")
	}
	if e.onTray != nil {
		e.onTray(false, 0, 0, 0)
	}

	e.mu.Lock()
	e.cmd, e.cancel = nil, nil
	e.mu.Unlock()

	if !stopped && e.onUnexpectedExit != nil {
		e.onUnexpectedExit(err)
	}
}
```

### 6.3 Парсер логов: детект готовности и человеческие ошибки

sing-box печатает уровень словом в начале строки, а факт успешного старта — строкой
`sing-box started (0.05s)`. На этом и строим переход в состояние «подключено» — ровно как сейчас
`Established DTLS connection` поднимает WireGuard.

```go
func (e *SingboxEngine) parseLogs(r io.Reader, p ConnectParams) {
	defer e.wg.Done()
	sc := bufio.NewScanner(r)
	sc.Buffer(make([]byte, 64*1024), 1024*1024)

	for sc.Scan() {
		line := sc.Text()

		if strings.Contains(line, "sing-box started") {
			e.mu.Lock()
			first := !e.started
			e.started = true
			e.mu.Unlock()
			if first {
				runtime.EventsEmit(e.appCtx, "state_changed", "running", "")
				runtime.EventsEmit(e.appCtx, "log", "INFO", "[SB] Туннель активен ✓")
				if e.onTray != nil {
					e.onTray(true, 0, 0, 0)
				}
				applyMSSClamp(singTunName, firstPositive(p.MTU, 1400)) // существующий netsh-код
				e.startStatsLoop()
				go func() {
					time.Sleep(2 * time.Second)
					if nat, err := CheckNATType(); err == nil && nat != nil {
						runtime.EventsEmit(e.appCtx, "nat_info", nat)
					}
				}()
			}
		}

		level, msg := splitSingboxLevel(line)
		if hint := singboxHint(msg); hint != "" {
			runtime.EventsEmit(e.appCtx, "log", "WARN", "[SB] "+hint)
		}
		runtime.EventsEmit(e.appCtx, "log", level, msg)
		if level == "ERROR" || level == "FATAL" {
			runtime.EventsEmit(e.appCtx, "error", msg)
		}
	}
}

// splitSingboxLevel вытаскивает уровень из строки вида
// "2026-09-12 12:00:00 INFO inbound/tun[tun-in]: started at fturn-tun".
func splitSingboxLevel(line string) (string, string) {
	for _, lv := range []string{"FATAL", "PANIC", "ERROR", "WARN", "INFO", "DEBUG", "TRACE"} {
		if i := strings.Index(line, " "+lv+" "); i >= 0 {
			out := lv
			if lv == "PANIC" {
				out = "FATAL"
			}
			return out, strings.TrimSpace(line[i+len(lv)+2:])
		}
	}
	return classifyLevel(line), line
}

// singboxHint превращает типовые ошибки ядра в подсказку на русском.
func singboxHint(msg string) string {
	low := strings.ToLower(msg)
	switch {
	case strings.Contains(low, "reality") && strings.Contains(low, "verify"):
		return "Reality: сервер отверг ClientHello — проверьте pbk/sid/sni и время на ПК"
	case strings.Contains(low, "certificate") || strings.Contains(low, "x509"):
		return "Ошибка TLS-сертификата: неверный sni или сбито системное время"
	case strings.Contains(low, "authentication failed") || strings.Contains(low, "invalid user"):
		return "Сервер не принял UUID/пароль — ссылка устарела"
	case strings.Contains(low, "operation not permitted") || strings.Contains(low, "access is denied"):
		return "Нет прав на создание TUN — запустите приложение от имени администратора"
	case strings.Contains(low, "configure tun interface"):
		return "Не удалось настроить TUN: конфликт с другим VPN или занято имя интерфейса"
	case strings.Contains(low, "i/o timeout") || strings.Contains(low, "connection reset"):
		return "Сервер не отвечает: возможна блокировка IP/порта провайдером"
	}
	return ""
}
```

### 6.4 Остановка процесса без зомби (важно для Windows)

Проблема, которую вы уже решали для `freeturnclient`: `Process.Kill()` не даёт ядру снять
маршруты, а при падении GUI дочерний процесс остаётся жить. Правильная схема на Windows —
**Job Object с `KILL_ON_JOB_CLOSE`** (гарантия, что ядро умрёт вместе с приложением, даже при
аварийном завершении) плюс попытка мягкой остановки через `CTRL_BREAK`:

```go
// backend/proc_windows.go
//go:build windows

package backend

import (
	"os/exec"
	"syscall"
	"unsafe"

	"golang.org/x/sys/windows"
)

var jobHandle windows.Handle

func initJob() {
	if jobHandle != 0 {
		return
	}
	h, err := windows.CreateJobObject(nil, nil)
	if err != nil {
		return
	}
	info := windows.JOBOBJECT_EXTENDED_LIMIT_INFORMATION{
		BasicLimitInformation: windows.JOBOBJECT_BASIC_LIMIT_INFORMATION{
			LimitFlags: windows.JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE,
		},
	}
	_, _ = windows.SetInformationJobObject(h,
		windows.JobObjectExtendedLimitInformation,
		uintptr(unsafe.Pointer(&info)), uint32(unsafe.Sizeof(info)))
	jobHandle = h
}

func prepareProcessGroup(cmd *exec.Cmd) {
	initJob()
	cmd.SysProcAttr = &syscall.SysProcAttr{
		HideWindow:    true,
		CreationFlags: windows.CREATE_NEW_PROCESS_GROUP | windows.CREATE_NO_WINDOW,
	}
}

// attachToJob вызывать сразу после cmd.Start().
func attachToJob(cmd *exec.Cmd) {
	if jobHandle == 0 || cmd.Process == nil {
		return
	}
	h, err := windows.OpenProcess(windows.PROCESS_SET_QUOTA|windows.PROCESS_TERMINATE,
		false, uint32(cmd.Process.Pid))
	if err == nil {
		_ = windows.AssignProcessToJobObject(jobHandle, h)
		_ = windows.CloseHandle(h)
	}
}

// signalStop просит процесс завершиться штатно.
func signalStop(cmd *exec.Cmd) error {
	if cmd.Process == nil {
		return nil
	}
	return windows.GenerateConsoleCtrlEvent(windows.CTRL_BREAK_EVENT, uint32(cmd.Process.Pid))
}
```

```go
func (e *SingboxEngine) Stop() {
	e.mu.Lock()
	e.userStopped = true
	cmd, cancel, exitChan := e.cmd, e.cancel, e.exitChan
	e.stopStatsLoopLocked()
	e.mu.Unlock()
	if cmd == nil {
		return
	}

	_ = signalStop(cmd)                    // 1) мягко
	select {                               // 2) ждём до 3 с
	case <-exitChan:
		return
	case <-time.After(3 * time.Second):
	}
	if cancel != nil {                     // 3) жёстко
		cancel()
	}
	if cmd.Process != nil {
		_ = cmd.Process.Kill()
	}
	<-exitChan
}
```

Для не-Windows — заглушки: `prepareProcessGroup` ставит `Setpgid: true`,
`signalStop` посылает `syscall.SIGTERM`.

### 6.5 Статистика: переиспользуем существующий цикл

`engine_stats.go` уже делает всё, что нужно, — просто параметризуем имя интерфейса и «потоки»:

```go
// было: rx, tx, err := getInterfaceBytes(wgIface)
// стало:
rx, tx, err := getInterfaceBytes(iface) // iface приходит из Engine.Iface()
```

Для sing-box «активные потоки» бессмысленны, поэтому в событие `stats` вместо
`active_streams/configured_max` отдаём `0/0`, а UI для sing-box-профиля просто не рисует
эту строку. Если хочется настоящей статистики по соединениям — Clash API (§10.2).

func (o *Orchestrator) startEngineOnly(p ConnectParams) error {
	prof, err := loadProfile(p.Profile)
	if err != nil { /* как было */ }

	// … перехват лога как было …

	var engine Engine
	switch prof.EngineKind() {
	case EngineSingbox:
		engine = NewSingboxEngine(o.appCtx, o.onTray, func(err error) { o.triggerReconnect() })
	default:
		engine = NewFreeturnEngine(o.appCtx, o.onTray, func(err error) { o.triggerReconnect() })
	}
	if err := engine.Start(p, prof); err != nil {
		runtime.EventsEmit(o.appCtx, "log", "ERROR", fmt.Sprintf("Ошибка запуска движка: %v", err))
		o.stopLogWriter()
		return err
	}
	o.mu.Lock()
	o.engine = engine // поле теперь типа Engine
	o.mu.Unlock()
	return nil
}
